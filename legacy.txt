    data = csv.reader(f)
    state = 0
    for row in data:
        # skip over the start
        if not state:
            state += 1
            continue

        try:
            row = [float(i.strip()) for i in row]
        except ValueError:
            state += 1
            continue

        # reading
        if state == 1:
            points.append(row)

            fweights1.append(row)

        for j in range(m):
